cmake_minimum_required(VERSION 3.21)
project(pico_tls_policy_tests LANGUAGES C CXX)
enable_testing()
if(NOT PICO_SDK_PATH)
    set(PICO_SDK_PATH "$ENV{PICO_SDK_PATH}")
endif()
set(MBEDTLS "${PICO_SDK_PATH}/lib/mbedtls")
if(NOT EXISTS "${MBEDTLS}/library/x509_crt.c")
    message(FATAL_ERROR "Set PICO_SDK_PATH to the same SDK used by the firmware")
endif()
file(GLOB MBEDTLS_SOURCES "${MBEDTLS}/library/*.c")
add_library(pico-native-mbedtls STATIC ${MBEDTLS_SOURCES})
target_include_directories(pico-native-mbedtls PUBLIC "${MBEDTLS}/include" ../../pico2w)
target_compile_definitions(pico-native-mbedtls PUBLIC MBEDTLS_CONFIG_FILE="${CMAKE_CURRENT_LIST_DIR}/../../pico2w/mbedtls_config.h")
add_executable(pico-tls-tests test_tls.cpp ../../src/platform/pico/time.cpp ../../src/lib/time_platform.cpp)
target_compile_features(pico-tls-tests PRIVATE cxx_std_20)
target_include_directories(pico-tls-tests PRIVATE fakes .. ../../include)
target_compile_definitions(pico-tls-tests PRIVATE NIO_ROOT="${CMAKE_CURRENT_LIST_DIR}/../..")
target_link_libraries(pico-tls-tests PRIVATE pico-native-mbedtls)
add_test(NAME pico-tls-tests COMMAND pico-tls-tests)
