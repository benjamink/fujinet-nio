cmake_minimum_required(VERSION 3.21)
project(pico_tls_adapter_tests LANGUAGES CXX)
enable_testing()
foreach(MODE default only additive default-badbundle only-badbundle additive-badbundle)
    string(REPLACE "-badbundle" "" TRUST ${MODE})
    add_executable(pico-tls-${MODE}-tests test_tls_adapter.cpp ../../src/platform/pico/tls_protocol.cpp)
    target_compile_features(pico-tls-${MODE}-tests PRIVATE cxx_std_20)
    target_include_directories(pico-tls-${MODE}-tests PRIVATE fakes ../pico/fakes .. ../../include)
    if(TRUST STREQUAL "only")
        target_compile_definitions(pico-tls-${MODE}-tests PRIVATE FN_HTTPS_TEST_CA=1)
    elseif(TRUST STREQUAL "additive")
        target_compile_definitions(pico-tls-${MODE}-tests PRIVATE FN_HTTPS_TEST_CA_ADDITIVE=1)
    endif()
    target_compile_options(pico-tls-${MODE}-tests PRIVATE -Wall -Wextra -Werror)
    if(MODE MATCHES "badbundle")
        # Only the fail-closed test makes sense with an invalid bundle.
        target_compile_definitions(pico-tls-${MODE}-tests PRIVATE FN_TEST_BAD_BUNDLE=1)
        add_test(NAME pico-tls-${MODE}-tests COMMAND pico-tls-${MODE}-tests -tc=*fails*closed*)
    else()
        add_test(NAME pico-tls-${MODE}-tests COMMAND pico-tls-${MODE}-tests -tce=*fails*closed*)
    endif()
endforeach()
