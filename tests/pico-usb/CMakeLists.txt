cmake_minimum_required(VERSION 3.21)
project(pico_usb_bridge_tests LANGUAGES CXX)
enable_testing()
find_package(Threads REQUIRED)
add_executable(pico-usb-bridge-tests test_bridge.cpp)
target_compile_features(pico-usb-bridge-tests PRIVATE cxx_std_20)
target_include_directories(pico-usb-bridge-tests PRIVATE ..)
target_compile_options(pico-usb-bridge-tests PRIVATE -Wall -Wextra -Werror)
target_link_libraries(pico-usb-bridge-tests PRIVATE Threads::Threads)
add_test(NAME pico-usb-bridge-tests COMMAND pico-usb-bridge-tests)
set_tests_properties(pico-usb-bridge-tests PROPERTIES TIMEOUT 20)
